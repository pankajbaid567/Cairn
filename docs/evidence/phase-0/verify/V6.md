```bash
docker exec cairn-db-1 psql -U postgres -X -A -t -d cairn \
  -c "select extname, extversion from pg_extension where extname in ('vector','pg_search','btree_gist') order by 1"
# btree_gist|1.8   pg_search|0.26.3   vector|0.8.6
```
